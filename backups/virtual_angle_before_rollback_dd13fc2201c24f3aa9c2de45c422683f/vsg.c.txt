#include "vsg.h"
#include "app/math/pi_controller.h"
#include <math.h>

volatile VSG_DEF VSG_CTRL = {0};
static PI_POS_DEF PI_QV;

static float WrapAngle(float angle)
{
    return atan2f(sinf(angle), cosf(angle));
}

static float WrapRelativeAngle(float angle)
{
    // No extra trig in the PCC feature path; phase inputs are already wrapped.
    if(angle > 3.14159265359f || angle < -3.14159265359f)
    {
        angle = fmodf(angle, TWO_PI);
        if(angle > 3.14159265359f) angle -= TWO_PI;
        if(angle < -3.14159265359f) angle += TWO_PI;
    }
    return angle;
}

static float LegacyAngleIncrement(float omega, float Ts)
{
    return (omega <= 0.015f && omega >= -0.015f) ? 0.0f : omega * Ts;
}

static void ResetAngleBase(volatile VSG_DEF *v)
{
    const VSG_ANGLE_BASE empty = {0};
    v->angle_base = empty;
}

static void ResetVirtualAngle(volatile VSG_DEF *v)
{
    v->virtual_angle_int = v->virtual_angle_rad = v->virtual_angle_deg = 0.0f;
    v->virtual_angle_legacy_int = v->virtual_angle_legacy_deg = 0.0f;
    v->virtual_angle_pcc_int = v->virtual_angle_pcc_deg = 0.0f;
    v->virtual_angle_phase_prev = v->virtual_angle_offset = 0.0f;
    v->virtual_angle_mode = v->virtual_angle_phase_valid = v->virtual_angle_valid = 0U;
    ResetAngleBase(v);
}

void VSG_Reset(volatile VSG_DEF *v)
{
    v->omega_m = v->omega_m_pu = v->omega_sync = 0.0f;
    ResetVirtualAngle(v);
    v->power_loop_cnt = 0U;
    PI_Pos_Reset(&PI_QV);
}

void VSG_Init(volatile VSG_DEF *v)
{
    VSG_Reset(v);
    v->omega0 = v->v_nom = v->v_delta_pu = 0.0f;
    PI_Pos_Init(&PI_QV, 0.0f, 0.0f, VSG_QV_LIMIT_PU, -VSG_QV_LIMIT_PU);
}

void VSG_EnterGrid(volatile VSG_DEF *v, const volatile VSG_PARAM *cfg, float omega0, float omegaSync, float voltage, float qError)
{
    VSG_Reset(v);
    v->omega0 = omega0;
    v->omega_sync = omegaSync;
    v->v_nom = voltage;
    v->v_delta_pu = 0.0f;
    PI_QV.kp = cfg->Kp_QV;
    PI_QV.ki = cfg->Ki_QV;
    PI_QV.ui = -cfg->Kp_QV * qError;
}

float VSG_PowerLoop(volatile VSG_DEF *v, const volatile VSG_PARAM *cfg, float pRef, float qRef, float pe, float qe, float voltageMax, float Ts)
{
    float deviationMax, voltage;
    v->omega_m_pu += (pRef - pe - cfg->dp_pu * v->omega_m_pu) * Ts / (2.0f * cfg->h);
    deviationMax = cfg->omega_limit / v->omega0;
    if(v->omega_m_pu > deviationMax) v->omega_m_pu = deviationMax;
    if(v->omega_m_pu < -deviationMax) v->omega_m_pu = -deviationMax;
    v->omega_m = v->omega_m_pu * v->omega0;
    PI_QV.kp = cfg->Kp_QV;
    PI_QV.ki = cfg->Ki_QV;
    v->v_delta_pu = PI_Pos_Cal(&PI_QV, qRef - qe, Ts);
    voltage = v->v_nom + v->v_delta_pu * SYS_V_BASE_V;
    if(voltageMax > cfg->v_max) voltageMax = cfg->v_max;
    if(voltage > voltageMax) voltage = voltageMax;
    if(voltage < cfg->v_min) voltage = cfg->v_min;
    return voltage;
}

void VSG_VirtualAngleStep(volatile VSG_DEF *v, float omegaP, float Ts, bool enabled)
{
    if(!enabled)
    {
        v->virtual_angle_int = v->virtual_angle_rad = v->virtual_angle_deg = 0.0f;
        return;
    }
    if(omegaP <= 0.015f && omegaP >= -0.015f) omegaP = 0.0f;
    v->virtual_angle_int += omegaP * Ts;
    v->virtual_angle_rad = atan2f(sinf(v->virtual_angle_int), cosf(v->virtual_angle_int));
    v->virtual_angle_deg = v->virtual_angle_rad * 57.29577951308232f;
}

void VSG_VirtualAngleUpdate(volatile VSG_DEF *v, float thetaInv, float thetaPcc, float Ts, bool enabled, bool pccValid, uint16_t mode)
{
    float phase, raw;
    if(!enabled)
    {
        ResetVirtualAngle(v);
        return;
    }
    mode = mode <= 2U ? mode : 0U;
    if(mode != v->virtual_angle_mode) ResetAngleBase(v);
    v->virtual_angle_legacy_int += LegacyAngleIncrement(v->omega_m, Ts);
    v->virtual_angle_legacy_deg = WrapAngle(v->virtual_angle_legacy_int) * 57.29577951308232f;
    pccValid = pccValid && isfinite(thetaInv) && isfinite(thetaPcc);
    if(pccValid)
    {
        phase = WrapRelativeAngle(thetaInv - thetaPcc);
        if(v->virtual_angle_phase_valid == 1U)
            v->virtual_angle_pcc_int += WrapRelativeAngle(phase - v->virtual_angle_phase_prev);
        // First/recovered PCC sample anchors phase without inventing an unobserved increment.
        v->virtual_angle_phase_prev = phase;
        v->virtual_angle_phase_valid = 1U;
    }
    else v->virtual_angle_phase_valid = 0U;
    v->virtual_angle_pcc_deg = WrapRelativeAngle(v->virtual_angle_pcc_int) * 57.29577951308232f;
    raw = (mode == 1U || (mode == 2U && v->angle_base.ready == 1U)) ? v->virtual_angle_pcc_int : v->virtual_angle_legacy_int;
    if(mode != v->virtual_angle_mode)
    {
        // Runtime mode changes preserve the current selected angle; reset restores original zero.
        v->virtual_angle_offset = v->virtual_angle_int - raw;
        v->virtual_angle_mode = mode;
    }
    v->virtual_angle_int = raw + v->virtual_angle_offset;
    v->virtual_angle_rad = mode == 0U ? WrapAngle(v->virtual_angle_int) : WrapRelativeAngle(v->virtual_angle_int);
    v->virtual_angle_deg = v->virtual_angle_rad * 57.29577951308232f;
    v->virtual_angle_valid = isfinite(v->virtual_angle_int) && (mode == 0U || pccValid) ? 1U : 0U;
}

bool VSG_VirtualAngleEstablish(volatile VSG_DEF *v, const volatile VIRTUAL_ANGLE_PARAM *cfg, float Ts, bool pllLocked, float frequency, float p, float q, float pRef, float qRef)
{
    volatile VSG_ANGLE_BASE *b = &v->angle_base;
    float pMean, qMean, angleMean, frequencyMean;
    bool stable;
    if(v->virtual_angle_mode != 2U || b->ready == 1U) return false;
    if(!pllLocked || v->virtual_angle_valid != 1U || !(pRef > 0.0f) || !(Ts > 0.0f)
       || !isfinite(frequency) || !isfinite(p) || !isfinite(q) || !isfinite(pRef) || !isfinite(qRef))
    {
        ResetAngleBase(v);
        return false;
    }
    // A new initial setpoint must first establish its own operating point.
    if(pRef != b->p_ref || qRef != b->q_ref)
    {
        ResetAngleBase(v);
        b->p_ref = pRef; b->q_ref = qRef;
    }
    b->elapsed_s += Ts;
    b->window_elapsed_s += Ts;
    b->p_sum += p; b->q_sum += q;
    b->angle_sum += v->virtual_angle_int;
    b->frequency_sum += frequency;
    b->samples++;
    if(b->window_elapsed_s < cfg->window_s) return false;
    pMean = b->p_sum / (float)b->samples;
    qMean = b->q_sum / (float)b->samples;
    angleMean = b->angle_sum / (float)b->samples;
    frequencyMean = b->frequency_sum / (float)b->samples;
    stable = b->have_window == 1U && b->elapsed_s >= cfg->establish_s
        && fabsf(frequencyMean - v->omega0 / TWO_PI) <= cfg->frequency_error_hz
        && fabsf(pMean - b->p_previous) <= cfg->power_change_pu
        && fabsf(qMean - b->q_previous) <= cfg->power_change_pu
        && fabsf(angleMean - b->angle_previous) * 57.29577951308232f <= cfg->angle_change_deg;
    b->stable_s = stable ? b->stable_s + b->window_elapsed_s : 0.0f;
    b->p_previous = pMean; b->q_previous = qMean; b->angle_previous = angleMean;
    b->have_window = 1U;
    b->window_elapsed_s = b->p_sum = b->q_sum = b->angle_sum = b->frequency_sum = 0.0f;
    b->samples = 0U;
    if(!stable || b->stable_s < cfg->stable_s) return false;
    // Preserve the complete legacy-established feature. Only future increments change reference.
    b->angle_rad = v->virtual_angle_int;
    b->pcc_rad = v->virtual_angle_pcc_int;
    v->virtual_angle_offset = b->angle_rad - b->pcc_rad;
    b->ready = 1U;
    return true;
}
